Código CSS

/* Esto son pseudocodigos */

/* Link no visitado */
a:link {
  color: white;
}

p {
  font-family: fantasy;
  color: white;
  font-size: 20px;
}

/* Link visitado */
a:visited {
  color: rgb(131, 131, 131);
}

/* Link si pasas el ratón por encima */
a:hover {
  color: rgb(255, 121, 208);
  font-size: 20px;
}

/* link si lo activas */
a:active {
  color: #0000FF;
}

Codigo CSS


/* Pseudocódigo para input */

/* Si es opcional ponte este color */
input:optional {
  background-color: lightgreen;
}

/* Si es obligatorio */
input:required {
  background-color: rgba(255, 0, 0, 0.396);
  border-color: rgba(255, 0, 0, 0.362);
}
